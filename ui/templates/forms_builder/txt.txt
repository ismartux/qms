def dashboard(request):
    templates = ChecklistTemplate.objects.all().order_by("-created_at")
    return render(
        request,
        "forms_builder/dashboard.html",

    )



froms_engine/admin_views/item.py

from django.shortcuts import render, redirect, get_object_or_404
from forms_engine.models import ChecklistSection, ChecklistItem, ChecklistRule


def manage_items(request, section_id):
    section = get_object_or_404(ChecklistSection, id=section_id)

    if request.method == "POST":
        item = ChecklistItem.objects.create(
            section=section,
            item_id=request.POST["item_id"],
            label=request.POST["label"],
            item_type=request.POST["item_type"],
            required=request.POST.get("required") == "on",
            order=request.POST["order"],
            severity_weight=request.POST.get("severity_weight", 1),
        )

        if request.POST.get("rule_type"):
            ChecklistRule.objects.create(
                item=item,
                rule_type=request.POST["rule_type"],
                condition_value=request.POST["condition_value"],
            )

        return redirect(request.path)

    return render(
        request,
        "forms_builder/item_manage.html",
        {
            "section": section,
            "items": section.items.all(),
        },
    )



froms_engine/admin_views/role.py

from django.shortcuts import render, redirect, get_object_or_404
from forms_engine.models import ChecklistTemplate, TemplateRole
from core.identity.models import UserScope
from core.identity.models import Role


def assign_roles(request, template_id):
    template = get_object_or_404(ChecklistTemplate, id=template_id)

    # 🔹 Get available roles from system
    available_roles = Role.objects.filter(is_active=True)

    if request.method == "POST":
        # Clear old roles
        TemplateRole.objects.filter(template=template).delete()

        selected_roles = request.POST.getlist("roles")

        for role_id in request.POST.getlist("roles"):
            TemplateRole.objects.create(
                template=template,
                role_id=role_id
            )

        return redirect(
            "create_version",
            template_id=template.id
        )

    assigned_roles = (
        template.role_assignments
        .values_list("role", flat=True)
    )

    return render(
        request,
        "forms_builder/role_assign.html",
        {
            "template": template,
            "available_roles": available_roles,
            "assigned_roles": assigned_roles,
        },
    )


froms_engine/admin_views/section.py

from django.shortcuts import render, redirect, get_object_or_404
from forms_engine.models import ChecklistVersion, ChecklistSection


def manage_sections(request, version_id):
    version = get_object_or_404(ChecklistVersion, id=version_id)

    if request.method == "POST":
        ChecklistSection.objects.create(
            version=version,
            title=request.POST["title"],
            order=request.POST["order"],
        )
        return redirect(request.path)

    return render(
        request,
        "forms_builder/section_manage.html",
        {
            "version": version,
            "sections": version.sections.all(),
        },
    )



froms_engine/admin_views/template.py

from django.shortcuts import render, redirect
from org.models import Plant, Product
from forms_engine.models import ChecklistTemplate


def create_template(request):
    if request.method == "POST":
        template = ChecklistTemplate.objects.create(
            code=request.POST["code"],
            name=request.POST["name"],
            description=request.POST.get("description", ""),
            is_active=request.POST.get("is_active") == "on",
            bitable_app_token=request.POST.get("bitable_app_token"),
            bitable_table_id=request.POST.get("bitable_table_id"),
        )

        template.plants.set(request.POST.getlist("plants"))
        template.products.set(request.POST.getlist("products"))

        return redirect(
            "assign_roles",
            template_id=template.id
        )

    return render(
        request,
        "forms_builder/template_create.html",
        {
            "plants": Plant.objects.all(),
            "products": Product.objects.all(),
        },
    )

  function toggleSidebar() {
    const sidebar = document.getElementById("sidebar");
    const backdrop = document.getElementById("sidebarBackdrop");

    if (window.innerWidth < 992) {
      sidebar.classList.toggle("mobile-open");
      backdrop.classList.toggle("show");
    } else {
      sidebar.classList.toggle("collapsed");
    }
  }

  function closeMobileSidebar() {
    document.getElementById("sidebar").classList.remove("mobile-open");
    document.getElementById("sidebarBackdrop").classList.remove("show");
  }
